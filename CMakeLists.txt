cmake_minimum_required(VERSION 3.12)
project(MultiExeProject)

include(ExternalProject)

# Function to add a custom executable
function(add_custom_executable name flags)
    ExternalProject_Add(
        ${name}
        SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}
        CONFIGURE_COMMAND ""
        BUILD_COMMAND make -f ${CMAKE_CURRENT_SOURCE_DIR}/Makefile KERNEL_NAME=${flags}
        INSTALL_COMMAND ""
        BUILD_IN_SOURCE 0
    )
endfunction()
